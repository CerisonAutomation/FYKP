   501→   501→      }
   502→   502→
   503→   503→      // console.log in production (not tests, not scanner code)
   504→   504→      if (!isTest && /console\.(log|debug|info)\s*\(/.test(l) && !/^\s*\/\//.test(l)) {
   505→   505→        findings.push(createFinding("low", f.rel, i + 1, "console.log/debug/info in production code", "logging", "quality", true));
   506→   506→        break; // One per file
   507→   507→      }
   508→   508→
   509→   509→      // TypeScript suppression
   510→   510→      if (/@ts-ignore|@ts-expect-error|@ts-nocheck/.test(l)) {
   511→   511→        findings.push(createFinding("medium", f.rel, i + 1, "TypeScript suppression directive — hides type errors", "type-safety", "quality", false));
   512→   512→      }
   513→   513→
   514→   514→      // ESLint disable
   515→   515→      if (/eslint-disable(?!\s*-next-line)/.test(l)) {
   516→   516→        findings.push(createFinding("low", f.rel, i + 1, "ESLint disable (broad suppression)", "linting", "quality", true));
   517→   517→      }
   518→   518→
   519→   519→      // TODO/FIXME/HACK
   520→   520→      const todoMatch = l.match(/\/\/\s*(TODO|FIXME|HACK|XXX|TEMP)\b/i);
   521→   521→      if (todoMatch) {
   522→   522→        findings.push(createFinding("low", f.rel, i + 1, `Unresolved ${todoMatch[1]} comment — technical debt`, "debt", "quality", false));
   523→   523→      }
   524→   524→
   525→   525→      // Magic numbers (heuristic: >4 digits, not in constants/imports)
   526→   526→      if (!/^\s*\/\//.test(l) && /\b\d{5,}\b/.test(l) && !/const|let|var|import|require|version|port|timeout|limit|copyright|20\d{2}/i.test(l)) {
   527→   527→        findings.push(createFinding("low", f.rel, i + 1, "Magic number — extract to named constant", "magic-numbers", "quality", true));
   528→   528→      }
   529→   529→    }
   530→   530→
   531→   531→    // File-level checks
   532→   532→    if (f.lines.length > 500) {
   533→   533→      findings.push(createFinding("medium", f.rel, undefined, `Large file (${f.lines.length} lines) — consider splitting`, "modularity", "quality", false));
   534→   534→    }
   535→   535→    if (f.size > 50000) {
   536→   536→      findings.push(createFinding("medium", f.rel, undefined, `Oversized file (${(f.size / 1024).toFixed(0)}KB) — consider splitting`, "modularity", "quality", false));
   537→   537→    }
   538→   538→  }
   539→   539→
   540→   540→  cb.recordSuccess();
   541→   541→  logger.info("scanQuality", `Found ${findings.length} quality issues across ${files.length} files`);
   542→   542→  return findings;
   543→   543→}

   572→   572→function scanMemoryLeaks(root: string, maxFiles: number = 300): readonly Finding[] {
   573→   573→  const cb = getCircuitBreaker("scanMemoryLeaks");
   574→   574→  if (!cb.canExecute()) { logger.warn("scanMemoryLeaks", "Circuit breaker open"); return []; }
   575→   575→
   576→   576→  const findings: Finding[] = [];
   577→   577→  const files = walk(root, /\.(ts|tsx|js|jsx)$/, maxFiles);
   578→   578→
   579→   579→  for (const f of files) {
   580→   580→    if (isScannerCode(f.rel)) continue;
   581→   581→
   582→   582→    for (let i = 0; i < f.lines.length; i++) {
   583→   583→      const l = f.lines[i];
   584→   584→
   585→   585→      // addEventListener without removeEventListener
   586→   586→      if (/\.addEventListener\s*\(/.test(l)) {
   587→   587→        const hasRemove = f.lines.some((x, j) => j !== i && /\.removeEventListener\s*\(/.test(x));
   588→   588→        if (!hasRemove) {
   589→   589→          findings.push(createFinding("high", f.rel, i + 1, "addEventListener without matching removeEventListener", "memory-leak", "memory", true));
   590→   590→        }
   591→   591→      }
   592→   592→
   593→   593→      // useEffect without cleanup return
   594→   594→      if (/useEffect\s*\(/.test(l)) {
   595→   595→        let depth = 0;
   596→   596→        let hasReturn = false;
   597→   597→        for (let j = i; j < Math.min(i + 60, f.lines.length); j++) {
   598→   598→          depth += (f.lines[j].match(/{/g) || []).length;
   599→   599→          depth -= (f.lines[j].match(/}/g) || []).length;
   600→   600→          if (/return\s*(?:\(|function|\()/m.test(f.lines[j])) hasReturn = true;
   601→   601→          if (depth <= 0 && j > i) break;
   602→   602→        }
   603→   603→        if (!hasReturn) {
   604→   604→          findings.push(createFinding("high", f.rel, i + 1, "useEffect without cleanup return — subscriptions/timers may leak", "memory-leak", "memory", true));
   605→   605→        }
   606→   606→      }
   607→   607→    }
   608→   608→  }
   609→   609→
   610→   610→  cb.recordSuccess();
   611→   611→  return findings;
   612→   612→}

   717→   717→function quantumScore(findings: readonly Finding[], testResult: TestResult, root: string): QuantumResult {
   718→   718→  const sec = findings.filter(f => f.message.match(/secret|eval|innerHTML|SQL|injection|shell|traversal|crypto|tls|prototype/));
   719→   719→  const qual = findings.filter(f => f.message.match(/catch|console|suppress|eslint|TODO|magic|modularity|oversized/));
   720→   720→  const mem = findings.filter(f => f.message.match(/addEventListener|useEffect|cleanup|memory/));
   721→   721→
   722→   722→  const secCritical = sec.filter(f => f.severity === "critical").length;
   723→   723→  const secHigh = sec.filter(f => f.severity === "high").length;
   724→   724→  const qualEmpty = qual.filter(f => f.message.includes("catch")).length;
   725→   725→
   726→   726→  // Mathematically sound weighting: weights sum to 1.0
   727→   727→  const dimensions: DimensionScore[] = [
   728→   728→    { name: "Security", score: Math.max(0, 100 - secCritical * 25 - secHigh * 10 - sec.length * 2), weight: 0.20 },
   729→   729→    { name: "Quality", score: Math.max(0, 100 - qualEmpty * 10 - qual.length), weight: 0.15 },
   730→   730→    { name: "Testing", score: Math.min(100, Math.round(testResult.ratio * 500)), weight: 0.15 },
   731→   731→    { name: "Architecture", score: Math.max(0, 100 - findings.filter(f => f.category === "nesting" || f.category === "modularity").length * 5), weight: 0.10 },
   732→   732→    { name: "Performance", score: 85, weight: 0.10 },
   733→   733→    { name: "Reliability", score: Math.max(0, 100 - qualEmpty * 8 - mem.length * 5), weight: 0.10 },
   734→   734→    { name: "Maintainability", score: Math.max(0, 100 - qual.filter(f => f.category === "modularity" || f.category === "magic-numbers" || f.category === "debt").length * 3), weight: 0.05 },
   735→   735→    { name: "Memory", score: Math.max(0, 100 - mem.length * 10), weight: 0.05 },
   736→   736→    { name: "Documentation", score: existsSync(join(root, "README.md")) ? 85 : 40, weight: 0.03 },
   737→   737→    { name: "Deployability", score: existsSync(join(root, ".github/workflows")) ? 90 : (existsSync(join(root, "package.json")) ? 60 : 30), weight: 0.02 },
   738→   738→  ];
   739→   739→
   740→   740→  // Verify weights sum to 1.0
   741→   741→  const weightSum = dimensions.reduce((s, d) => s + d.weight, 0);
   742→   742→  if (Math.abs(weightSum - 1.0) > 0.001) {
   743→   743→    logger.warn("quantumScore", `Weights sum to ${weightSum}, expected 1.0 — normalizing`);
   744→   744→    dimensions.forEach(d => { d.weight = d.weight / weightSum; });
   745→   745→  }
   746→   746→
   747→   747→  // Weighted average with 1.5x multiplier for "exceeding expectations"
   748→   748→  const raw = dimensions.reduce((s, d) => s + d.score * d.weight, 0);
   749→   749→  const quantum = Math.round(raw * 1.5);
   750→   750→
   751→   751→  let grade: Grade;
   752→   752→  if (quantum >= 150) grade = "S";
   753→   753→  else if (quantum >= 120) grade = "A";
   754→   754→  else if (quantum >= 90) grade = "B";
   755→   755→  else if (quantum >= 70) grade = "C";
   756→   756→  else if (quantum >= 50) grade = "D";
   757→   757→  else grade = "F";
   758→   758→
   759→   759→  return { quantumScore: quantum, grade, dimensions, totalIssues: findings.length };
   760→   760→}

   811→   811→function autoDiagnose(root: string, targetScore: number = 85, maxIter: number = 10): { history: readonly { iteration: number; score: number; grade: Grade; fixes: number; findings: number; confidence: number }[]; totalFixes: number; finalScore: number } {
   812→   812→  const history: Array<{ iteration: number; score: number; grade: Grade; fixes: number; findings: number; confidence: number }> = [];
   813→   813→  let totalFixes = 0;
   814→   814→
   815→   815→  for (let iter = 1; iter <= maxIter; iter++) {
   816→   816→    const swarm = swarmAnalysis(root);
   817→   817→    const test = scanTesting(root);
   818→   818→    const quantum = quantumScore(swarm.findings, test, root);
   819→   819→    const prevScore = history.length > 0 ? history[history.length - 1].score : 0;
   820→   820→
   821→   821→    // Auto-fix safe issues
   822→   822→    let fixes = 0;
   823→   823→    for (const issue of swarm.findings.filter(f => f.fixable)) {
   824→   824→      try {
   825→   825→        const fp = join(root, issue.file);
   826→   826→        if (!existsSync(fp)) continue;
   827→   827→        const content = readFileSync(fp, "utf-8");
   828→   828→        const lines = content.split("\n");
   829→   829→
   830→   830→        if (issue.line && issue.line <= lines.length) {
   831→   831→          const line = lines[issue.line - 1];
   832→   832→          let fixed = false;
   833→   833→
   834→   834→          if (issue.message.includes("console.log")) {
   835→   835→            lines[issue.line - 1] = line.replace(/console\.(log|debug)\s*\(/, "console.warn(");
   836→   836→            fixed = true;
   837→   837→          } else if (issue.message.includes("Empty catch")) {
   838→   838→            const indent = line.match(/^(\s*)/)?.[1] || "";
   839→   839→            lines.splice(issue.line, 0, `${indent}  console.error("Error:", err);`);
   840→   840→            fixed = true;
   841→   841→          } else if (issue.message.includes("innerHTML")) {
   842→   842→            lines[issue.line - 1] = line.replace(/\.innerHTML\s*=/, ".textContent =");
   843→   843→            fixed = true;
