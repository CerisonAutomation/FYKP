   108→   108→            key={i}

   111→   111→            transition={{ duration: 0.8 + i * 0.05, repeat: Infinity, ease: 'easeInOut' }}
   112→   112→            style={{ height: `${h}%`, transformOrigin: 'bottom' }}
   113→   113→          />
   114→   114→        );
   115→   115→      })}
