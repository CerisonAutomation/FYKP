   175→function timeAgo(dateStr: string): string {
   176→  try { return formatDistanceToNow(new Date(dateStr), { addSuffix: true }); } catch { return ''; }
   177→}
   178→
   179→function fmtDate(dateStr: string): string {
   180→  try { return format(new Date(dateStr), 'MMM d, yyyy h:mm a'); } catch { return dateStr; }
   181→}

   183→function fmtDateShort(dateStr: string): string {
   184→  try { return format(new Date(dateStr), 'MMM d'); } catch { return dateStr; }
   185→}

   191→function truncate(str: string | null | undefined, len: number): string {
   192→  if (!str) return '';
   193→  return str.length > len ? str.slice(0, len) + '...' : str;
   194→}

   196→function getLastSeenText(user: User): string {

   199→  return 'Offline';
