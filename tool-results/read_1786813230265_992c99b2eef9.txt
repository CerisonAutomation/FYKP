   208→type LogLevel = "debug" | "info" | "warn" | "error";

   210→interface LogEntry {
   211→  readonly timestamp: string;
   212→  readonly level: LogLevel;
   213→  readonly component: string;
   214→  readonly message: string;
   215→  readonly data?: Record<string, unknown>;

   217→
   218→class AuditLogger {
   219→  private entries: LogEntry[] = [];
   220→  private maxEntries = 1000;
   221→
   222→  log(level: LogLevel, component: string, message: string, data?: Record<string, unknown>): void {
   223→    const entry: LogEntry = { timestamp: new Date().toISOString(), level, component, message, data };
   224→    this.entries.push(entry);
   225→    if (this.entries.length > this.maxEntries) this.entries.splice(0, this.entries.length - 500);
   226→    // Structured output to stderr (MCP convention)
   227→    const prefix = level === "error" ? "❌" : level === "warn" ? "⚠️" : level === "info" ? "✅" : "🔍";
   228→    console.error(`[repo-intel] ${prefix} [${component}] ${message}`);
   229→  }
   230→
   231→  debug(component: string, message: string, data?: Record<string, unknown>): void { this.log("debug", component, message, data); }
   232→  info(component: string, message: string, data?: Record<string, unknown>): void { this.log("info", component, message, data); }
   233→  warn(component: string, message: string, data?: Record<string, unknown>): void { this.log("warn", component, message, data); }
   234→  error(component: string, message: string, data?: Record<string, unknown>): void { this.log("error", component, message, data); }
   235→
   236→  getEntries(filter?: { level?: LogLevel; component?: string; limit?: number }): readonly LogEntry[] {
   237→    let result = this.entries;
   238→    if (filter?.level) result = result.filter(e => e.level === filter.level);
   239→    if (filter?.component) result = result.filter(e => e.component === filter.component);
   240→    if (filter?.limit) result = result.slice(-filter.limit);
   241→    return result;
   242→  }
   243→}

   251→class CircuitBreaker {
   252→  private failures = 0;
   253→  private lastFailure = 0;
   254→  private state: "closed" | "open" | "half-open" = "closed";
   255→  private readonly threshold: number;
   256→  private readonly resetTimeout: number;
   257→
   258→  constructor(threshold = 5, resetTimeout = 60000) {
   259→    this.threshold = threshold;
   260→    this.resetTimeout = resetTimeout;
   261→  }
   262→
   263→  canExecute(): boolean {
   264→    if (this.state === "closed") return true;
   265→    if (this.state === "open" && Date.now() - this.lastFailure > this.resetTimeout) {
   266→      this.state = "half-open";
   267→      return true;
   268→    }
   269→    return this.state === "half-open";
   270→  }
   271→
   272→  recordSuccess(): void { this.failures = 0; this.state = "closed"; }
   273→  recordFailure(): void { this.failures++; this.lastFailure = Date.now(); if (this.failures >= this.threshold) this.state = "open"; }
   274→  getState(): string { return this.state; }
   275→  reset(): void { this.failures = 0; this.state = "closed"; }
   276→}

   278→const circuitBreakers = new Map<string, CircuitBreaker>();
   279→function getCircuitBreaker(name: string): CircuitBreaker {
   280→  if (!circuitBreakers.has(name)) circuitBreakers.set(name, new CircuitBreaker());
   281→  return circuitBreakers.get(name)!;

   284→// ═══════════════════════════════════════════════════════════════════════════════
   285→// SECTION 4: MEMORY-SAFE FILE WALKER
   286→// ═══════════════════════════════════════════════════════════════════════════════
   287→
   288→const EXCLUDE = new Set(["node_modules", ".git", ".next", "dist", "build", "__pycache__", "coverage", ".dyad", ".claude", ".github", ".repo-intel"]);
   289→const MAX_FILE_SIZE = 1_000_000; // 1MB max per file
   290→const MAX_TOTAL_BYTES = 50_000_000; // 50MB total scan limit
   291→
   292→function walk(root: string, filter: RegExp, maxFiles: number = 500, maxDepth: number = 6): FileData[] {
   293→  const files: FileData[] = [];
   294→  let totalBytes = 0;
   295→
   296→  function _walk(dir: string, depth: number): void {
   297→    if (depth > maxDepth || files.length >= maxFiles || totalBytes >= MAX_TOTAL_BYTES) return;
   298→    let entries: string[];
   299→    try { entries = readdirSync(dir); } catch { return; }
   300→
   301→    for (const entry of entries) {
   302→      if (files.length >= maxFiles || totalBytes >= MAX_TOTAL_BYTES) return;
   303→      if (EXCLUDE.has(entry) || entry.startsWith(".")) continue;
   304→
   305→      const fp = join(dir, entry);
   306→      let st: ReturnType<typeof statSync>;
   307→      try { st = statSync(fp); } catch { continue; }
   308→
   309→      if (st.isDirectory()) {
   310→        _walk(fp, depth + 1);
   311→      } else if (st.isFile() && st.size <= MAX_FILE_SIZE && filter.test(entry)) {
   312→        try {
   313→          const content = readFileSync(fp, "utf-8");
   314→          totalBytes += content.length;
   315→          const ext = entry.split(".").pop() || "unknown";
   316→          files.push({ path: fp, rel: relative(root, fp), content, lines: Object.freeze(content.split("\n")), size: st.size, ext });
   317→        } catch { /* Skip unreadable files */ }
   318→      }
   319→    }
   320→  }
   321→
   322→  _walk(root, 0);
   323→  logger.info("walk", `Scanned ${files.length} files (${(totalBytes / 1024).toFixed(0)}KB)`);
   324→  return files;
   325→}

   334→function sanitizeCommand(cmd: string): string {
   335→  // Remove null bytes, dangerous characters, and truncate
   336→  return cmd.replace(/\0/g, "").replace(/[;&|`$(){}[\]!#~]/g, "").substring(0, MAX_CMD_LENGTH);

   339→function safeExec(root: string, name: string, cmd: string, timeout = 60000): VerifyCheck {
   340→  const cb = getCircuitBreaker(`exec:${name}`);
   341→  if (!cb.canExecute()) {
   342→    logger.warn("exec", `Circuit breaker open for ${name}, skipping`);
   343→    return { name, status: "skip", output: "Circuit breaker open — too many recent failures" };
   344→  }
   345→
   346→  const sanitized = sanitizeCommand(cmd);
   347→  try {
   348→    execFileSync("sh", ["-c", sanitized], { cwd: root, timeout, stdio: "pipe", env: { ...process.env, NODE_OPTIONS: "" } });
   349→    cb.recordSuccess();
   350→    return { name, status: "pass", command: sanitized };
   351→  } catch (err: any) {
   352→    cb.recordFailure();
   353→    const output = (err.stderr || err.stdout || err.message || "").substring(0, 500);
   354→    logger.error("exec", `${name} failed`, { exitCode: err.status, output: output.substring(0, 200) });
   355→    return { name, status: "fail", command: sanitized, output };
   356→  }
   357→}

   382→interface CritiqueResult {
   383→  readonly validated: readonly Finding[];
   384→  readonly falsePositives: number;
   385→  readonly confidence: number;
   386→  readonly deduplicated: number;
   387→}

   389→function critiqueFindings(findings: readonly Finding[]): CritiqueResult {
   390→  const validated: Finding[] = [];
   391→  let falsePositives = 0;
   392→  let deduplicated = 0;
   393→  const seen = new Set<string>();
   394→
   395→  for (const f of findings) {
   396→    // Skip test file false positives
   397→    if (f.severity === "critical" && f.file.match(/\.(test|spec|mock|fixture)\./)) { falsePositives++; continue; }
   398→    // Skip vendor code
   399→    if (f.file.includes("vendor/")) { falsePositives++; continue; }
   400→    // Deduplicate
   401→    const key = `${f.file}:${f.line}:${f.message}`;
   402→    if (seen.has(key)) { deduplicated++; continue; }
   403→    seen.add(key);
   404→    validated.push(f);
   405→  }
   406→
   407→  const confidence = findings.length > 0 ? Math.round((validated.length / findings.length) * 100) : 100;
   408→  logger.info("critique", `Validated ${validated.length}/${findings.length} findings (FP: ${falsePositives}, Deduped: ${deduplicated}, Confidence: ${confidence}%)`);
   409→  return { validated, falsePositives, confidence, deduplicated };
   410→}

   416→function scanSecurity(root: string, maxFiles: number = 500): readonly Finding[] {
   417→  const cb = getCircuitBreaker("scanSecurity");
   418→  if (!cb.canExecute()) { logger.warn("scanSecurity", "Circuit breaker open"); return []; }
   419→
   420→  const findings: Finding[] = [];
   421→  const files = walk(root, /\.(ts|tsx|js|jsx|py|go|rs|java|rb|php)$/, maxFiles);
   422→
   423→  for (const f of files) {
   424→    if (isScannerCode(f.rel)) continue; // Skip scanner code itself
   425→
   426→    for (let i = 0; i < f.lines.length; i++) {
   427→      const l = f.lines[i];
   428→
   429→      // Hardcoded secrets (improved regex — covers more patterns)
   430→      if (/(?:password|secret|api[_-]?key|token|credentials?|auth[_-]?key)\s*[:=]\s*["'`][A-Za-z0-9+/=_-]{16,}["'`]/i.test(l)) {
   431→        findings.push(createFinding("critical", f.rel, i + 1, "Hardcoded secret/credential detected", "secrets", "security", false));
   432→      }
   433→
   434→      // Code injection (eval/exec/Function)
   435→      if (/\b(?:eval|exec|Function|setTimeout|setInterval)\s*\(\s*["'`]/.test(l)) {
   436→        findings.push(createFinding("critical", f.rel, i + 1, "Dynamic code execution from string literal", "injection", "security", false));
   437→      }
   438→
   439→      // innerHTML XSS
   440→      if (/\.innerHTML\s*=/.test(l) && !/^\s*\/\//.test(l)) {
   441→        findings.push(createFinding("high", f.rel, i + 1, "innerHTML assignment — XSS vulnerability", "xss", "security", true));
   442→      }
   443→
   444→      // SQL injection via template literals
   445→      if (/`[^`]*\$\{[^}]*\}[^`]*(?:SELECT|INSERT|UPDATE|DELETE|DROP)/i.test(l)) {
   446→        findings.push(createFinding("critical", f.rel, i + 1, "SQL injection via template literal interpolation", "injection", "security", false));
   447→      }
   448→
   449→      // Shell injection
   450→      if (/(?:execSync|exec|spawn|system|child_process)\s*\(/.test(l) && /shell\s*[:=]\s*true/.test(l)) {
   451→        findings.push(createFinding("high", f.rel, i + 1, "Shell injection risk — shell: true with user input", "injection", "security", false));
   452→      }
   453→
   454→      // Path traversal
   455→      if (/(?:path\.join|path\.resolve)\s*\(.*(?:req|request|params|query|body|input)/.test(l)) {
   456→        findings.push(createFinding("medium", f.rel, i + 1, "Potential path traversal — user input in path construction", "traversal", "security", true));
   457→      }
   458→
   459→      // Weak hashing
   460→      if (/\b(?:md5|sha1)\s*\(/.test(l) && !/^\s*\/\//.test(l)) {
   461→        findings.push(createFinding("medium", f.rel, i + 1, "Weak hashing algorithm (md5/sha1) — use sha256+ or bcrypt", "crypto", "security", true));
   462→      }
   463→
   464→      // TLS verification disabled
   465→      if (/verify\s*[:=]\s*false/i.test(l) && !/^\s*\/\//.test(l)) {
   466→        findings.push(createFinding("high", f.rel, i + 1, "TLS verification disabled — MITM risk", "tls", "security", true));
   467→      }
   468→
   469→      // Prototype pollution
   470→      if (/__proto__|constructor\[/.test(l)) {
   471→        findings.push(createFinding("critical", f.rel, i + 1, "Prototype pollution vector", "injection", "security", false));
   472→      }
   473→    }
   474→  }
   475→
   476→  cb.recordSuccess();
   477→  logger.info("scanSecurity", `Found ${findings.length} security issues across ${files.length} files`);
   478→  return findings;
   479→}

   481→function scanQuality(root: string, maxFiles: number = 500): readonly Finding[] {
   482→  const cb = getCircuitBreaker("scanQuality");
   483→  if (!cb.canExecute()) { logger.warn("scanQuality", "Circuit breaker open"); return []; }
   484→
   485→  const findings: Finding[] = [];
   486→  const files = walk(root, /\.(ts|tsx|js|jsx)$/, maxFiles);
   487→
   488→  for (const f of files) {
   489→    const isTest = /\.(test|spec)\./.test(f.rel);
   490→    if (isScannerCode(f.rel)) continue;
   491→
   492→    for (let i = 0; i < f.lines.length; i++) {
   493→      const l = f.lines[i];
   494→
   495→      // Empty catch blocks (improved detection)
   496→      if (/catch\s*\(/.test(l)) {
   497→        const nextBlock = f.lines.slice(i + 1, i + 5).join("");
   498→        if (/^\s*\{\s*\}/.test(nextBlock) || /^\s*\{\s*\/\*.*\*\/\s*\}/.test(nextBlock) || /^\s*\{\s*\/\/.*\}/.test(nextBlock)) {
   499→          findings.push(createFinding("high", f.rel, i + 1, "Empty catch block — errors silently swallowed", "error-handling", "quality", true));
   500→        }
   501→      }
   502→
   503→      // console.log in production (not tests, not scanner code)
   504→      if (!isTest && /console\.(log|debug|info)\s*\(/.test(l) && !/^\s*\/\//.test(l)) {
   505→        findings.push(createFinding("low", f.rel, i + 1, "console.log/debug/info in production code", "logging", "quality", true));
   506→        break; // One per file
   507→      }
   508→
   509→      // TypeScript suppression
   510→      if (/@ts-ignore|@ts-expect-error|@ts-nocheck/.test(l)) {
   511→        findings.push(createFinding("medium", f.rel, i + 1, "TypeScript suppression directive — hides type errors", "type-safety", "quality", false));
   512→      }
   513→
   514→      // ESLint disable
   515→      if (/eslint-disable(?!\s*-next-line)/.test(l)) {
   516→        findings.push(createFinding("low", f.rel, i + 1, "ESLint disable (broad suppression)", "linting", "quality", true));
   517→      }
   518→
   519→      // TODO/FIXME/HACK
   520→      const todoMatch = l.match(/\/\/\s*(TODO|FIXME|HACK|XXX|TEMP)\b/i);
   521→      if (todoMatch) {
   522→        findings.push(createFinding("low", f.rel, i + 1, `Unresolved ${todoMatch[1]} comment — technical debt`, "debt", "quality", false));
   523→      }
   524→
   525→      // Magic numbers (heuristic: >4 digits, not in constants/imports)
   526→      if (!/^\s*\/\//.test(l) && /\b\d{5,}\b/.test(l) && !/const|let|var|import|require|version|port|timeout|limit|copyright|20\d{2}/i.test(l)) {
   527→        findings.push(createFinding("low", f.rel, i + 1, "Magic number — extract to named constant", "magic-numbers", "quality", true));
   528→      }
   529→    }
   530→
   531→    // File-level checks
   532→    if (f.lines.length > 500) {
   533→      findings.push(createFinding("medium", f.rel, undefined, `Large file (${f.lines.length} lines) — consider splitting`, "modularity", "quality", false));
   534→    }
   535→    if (f.size > 50000) {
   536→      findings.push(createFinding("medium", f.rel, undefined, `Oversized file (${(f.size / 1024).toFixed(0)}KB) — consider splitting`, "modularity", "quality", false));
   537→    }
   538→  }
   539→
   540→  cb.recordSuccess();
   541→  logger.info("scanQuality", `Found ${findings.length} quality issues across ${files.length} files`);
   542→  return findings;
   543→}

   545→function scanArchitecture(root: string): readonly Finding[] {
   546→  const cb = getCircuitBreaker("scanArchitecture");
   547→  if (!cb.canExecute()) { logger.warn("scanArchitecture", "Circuit breaker open"); return []; }
   548→
   549→  const findings: Finding[] = [];
   550→
   551→  // Deep nesting detection
   552→  let deepDirs = 0;
   553→  function checkDepth(dir: string, d: number): void {
   554→    if (d > 7) { deepDirs++; return; }
   555→    let entries: string[];
   556→    try { entries = readdirSync(dir); } catch { return; }
   557→    for (const entry of entries) {
   558→      if (EXCLUDE.has(entry) || entry.startsWith(".")) continue;
   559→      const fp = join(dir, entry);
   560→      try { if (statSync(fp).isDirectory()) checkDepth(fp, d + 1); } catch {}
   561→    }
   562→  }
   563→  checkDepth(root, 0);
   564→  if (deepDirs > 0) {
   565→    findings.push(createFinding("medium", "(directory structure)", undefined, `${deepDirs} directories deeper than 7 levels — consider flattening`, "nesting", "architecture", false));
   566→  }
   567→
   568→  cb.recordSuccess();
   569→  return findings;
   570→}

   572→function scanMemoryLeaks(root: string, maxFiles: number = 300): readonly Finding[] {
   573→  const cb = getCircuitBreaker("scanMemoryLeaks");
   574→  if (!cb.canExecute()) { logger.warn("scanMemoryLeaks", "Circuit breaker open"); return []; }
   575→
   576→  const findings: Finding[] = [];
   577→  const files = walk(root, /\.(ts|tsx|js|jsx)$/, maxFiles);
   578→
   579→  for (const f of files) {
   580→    if (isScannerCode(f.rel)) continue;
   581→
   582→    for (let i = 0; i < f.lines.length; i++) {
   583→      const l = f.lines[i];
   584→
   585→      // addEventListener without removeEventListener
   586→      if (/\.addEventListener\s*\(/.test(l)) {
   587→        const hasRemove = f.lines.some((x, j) => j !== i && /\.removeEventListener\s*\(/.test(x));
   588→        if (!hasRemove) {
   589→          findings.push(createFinding("high", f.rel, i + 1, "addEventListener without matching removeEventListener", "memory-leak", "memory", true));
   590→        }
   591→      }
   592→
   593→      // useEffect without cleanup return
   594→      if (/useEffect\s*\(/.test(l)) {
   595→        let depth = 0;
   596→        let hasReturn = false;
   597→        for (let j = i; j < Math.min(i + 60, f.lines.length); j++) {
   598→          depth += (f.lines[j].match(/{/g) || []).length;
   599→          depth -= (f.lines[j].match(/}/g) || []).length;
   600→          if (/return\s*(?:\(|function|\()/m.test(f.lines[j])) hasReturn = true;
   601→          if (depth <= 0 && j > i) break;
   602→        }
   603→        if (!hasReturn) {
   604→          findings.push(createFinding("high", f.rel, i + 1, "useEffect without cleanup return — subscriptions/timers may leak", "memory-leak", "memory", true));
   605→        }
   606→      }
   607→    }
   608→  }
   609→
   610→  cb.recordSuccess();
   611→  return findings;
   612→}

   615→  const cb = getCircuitBreaker("scanCleanup");
   616→  if (!cb.canExecute()) { logger.warn("scanCleanup", "Circuit breaker open"); return []; }
   617→
   618→  const findings: Finding[] = [];
   619→  const files = walk(root, /\.(ts|tsx|js|jsx|json|md|yaml|yml|css|html)$/, maxFiles);
   620→
   621→  for (const f of files) {
   622→    if (f.content.includes("@generated") || f.content.includes("AUTO GENERATED") || f.content.includes("do not edit")) {
   623→      findings.push(createFinding("medium", f.rel, undefined, "Auto-generated file in source tree", "cleanup", "cleanup", true));

   626→      findings.push(createFinding("low", f.rel, undefined, "Empty or comment-only file", "cleanup", "cleanup", true));

   628→    if (f.rel.match(/\.(bak|backup|old|orig|save|tmp)$/) && statSync(f.path).mtimeMs < Date.now() - 30 * 86400000) {
   629→      findings.push(createFinding("low", f.rel, undefined, "Stale backup file (>30 days old)", "cleanup", "cleanup", true));
   630→    }
   631→    const todoCount = (f.content.match(/(?:TODO|FIXME|HACK|XXX)/gi) || []).length;
   632→    if (todoCount > 5) {
   633→      findings.push(createFinding("medium", f.rel, undefined, `${todoCount} TODO/FIXME comments — accumulated tech debt`, "debt", "cleanup", false));
   634→    }
   635→  }
   636→
   637→  cb.recordSuccess();
   638→  return findings;

   660→function scanTesting(root: string): TestResult {
   661→  const findings: Finding[] = [];
   662→  const testFiles = walk(root, /\.(test|spec)\.(ts|tsx|js|jsx)$/, 500);
   663→  const sourceFiles = walk(root, /\.(ts|tsx|js|jsx)$/, 2000).filter(f => ! /\.(test|spec)\./.test(f.rel));
   664→  const ratio = sourceFiles.length > 0 ? testFiles.length / sourceFiles.length : 0;
   665→
   666→  for (const tf of testFiles) {
   667→    const hasAssert = tf.lines.some(l => /(?:expect|assert|should|\.toBe|\.toEqual|\.toMatch|\.toThrow|\.toHaveBeenCalled|\.resolves|\.rejects)/.test(l));
   668→    if (!hasAssert && tf.lines.length > 10) {
   669→      findings.push(createFinding("high", tf.rel, undefined, "Test file has no assertions — passes by not throwing", "test-quality", "testing", false));
   670→    }
   671→  }
   672→
   673→  for (const tf of testFiles) {
   674→    for (const l of tf.lines) {
   675→      if (/\b(xit|xtest|\.skip|pending|@ignore|@Disabled|\.only)\b/.test(l)) {
   676→        findings.push(createFinding("medium", tf.rel, undefined, "Test is disabled/skipped/only", "test-coverage", "testing", false));
   677→        break;
   678→      }
   679→    }
   680→  }
   681→
   682→  return { testFiles: testFiles.length, sourceFiles: sourceFiles.length, ratio, findings };
   683→}

   717→function quantumScore(findings: readonly Finding[], testResult: TestResult, root: string): QuantumResult {
   718→  const sec = findings.filter(f => f.message.match(/secret|eval|innerHTML|SQL|injection|shell|traversal|crypto|tls|prototype/));
   719→  const qual = findings.filter(f => f.message.match(/catch|console|suppress|eslint|TODO|magic|modularity|oversized/));
   720→  const mem = findings.filter(f => f.message.match(/addEventListener|useEffect|cleanup|memory/));
   721→
   722→  const secCritical = sec.filter(f => f.severity === "critical").length;
   723→  const secHigh = sec.filter(f => f.severity === "high").length;
   724→  const qualEmpty = qual.filter(f => f.message.includes("catch")).length;
   725→
   726→  // Mathematically sound weighting: weights sum to 1.0
   727→  const dimensions: DimensionScore[] = [
   728→    { name: "Security", score: Math.max(0, 100 - secCritical * 25 - secHigh * 10 - sec.length * 2), weight: 0.20 },
   729→    { name: "Quality", score: Math.max(0, 100 - qualEmpty * 10 - qual.length), weight: 0.15 },
   730→    { name: "Testing", score: Math.min(100, Math.round(testResult.ratio * 500)), weight: 0.15 },
   731→    { name: "Architecture", score: Math.max(0, 100 - findings.filter(f => f.category === "nesting" || f.category === "modularity").length * 5), weight: 0.10 },
   732→    { name: "Performance", score: 85, weight: 0.10 },
   733→    { name: "Reliability", score: Math.max(0, 100 - qualEmpty * 8 - mem.length * 5), weight: 0.10 },
   734→    { name: "Maintainability", score: Math.max(0, 100 - qual.filter(f => f.category === "modularity" || f.category === "magic-numbers" || f.category === "debt").length * 3), weight: 0.05 },
   735→    { name: "Memory", score: Math.max(0, 100 - mem.length * 10), weight: 0.05 },
   736→    { name: "Documentation", score: existsSync(join(root, "README.md")) ? 85 : 40, weight: 0.03 },
   737→    { name: "Deployability", score: existsSync(join(root, ".github/workflows")) ? 90 : (existsSync(join(root, "package.json")) ? 60 : 30), weight: 0.02 },
   738→  ];
   739→
   740→  // Verify weights sum to 1.0
   741→  const weightSum = dimensions.reduce((s, d) => s + d.weight, 0);
   742→  if (Math.abs(weightSum - 1.0) > 0.001) {
   743→    logger.warn("quantumScore", `Weights sum to ${weightSum}, expected 1.0 — normalizing`);
   744→    dimensions.forEach(d => { d.weight = d.weight / weightSum; });
   745→  }
   746→
   747→  // Weighted average with 1.5x multiplier for "exceeding expectations"
   748→  const raw = dimensions.reduce((s, d) => s + d.score * d.weight, 0);
   749→  const quantum = Math.round(raw * 1.5);
   750→
   751→  let grade: Grade;
   752→  if (quantum >= 150) grade = "S";
   753→  else if (quantum >= 120) grade = "A";
   754→  else if (quantum >= 90) grade = "B";
   755→  else if (quantum >= 70) grade = "C";
   756→  else if (quantum >= 50) grade = "D";
   757→  else grade = "F";
   758→
   759→  return { quantumScore: quantum, grade, dimensions, totalIssues: findings.length };
   760→}

   767→  const sec = scanSecurity(root, 300);

   777→
   778→  const byAgent: Record<string, number> = {};
   779→  const bySeverity: Record<string, number> = {};
   780→  for (const f of validated) {
   781→    byAgent[f.agent] = (byAgent[f.agent] ?? 0) + 1;
   782→    bySeverity[f.severity] = (bySeverity[f.severity] ?? 0) + 1;
   783→  }
   784→
   785→  // Cross-agent correlations
   786→  const correlations: string[] = [];

   788→  const memFiles = new Set(mem.map(f => f.file));

   790→  if (overlap.length > 0) correlations.push(`${overlap.length} files have both security and memory issues`);
   791→
   792→  return { findings: validated, falsePositives, confidence, byAgent, bySeverity, correlations, testResult: test };
   793→}

   811→function autoDiagnose(root: string, targetScore: number = 85, maxIter: number = 10): { history: readonly { iteration: number; score: number; grade: Grade; fixes: number; findings: number; confidence: number }[]; totalFixes: number; finalScore: number } {
   812→  const history: Array<{ iteration: number; score: number; grade: Grade; fixes: number; findings: number; confidence: number }> = [];
   813→  let totalFixes = 0;
   814→
   815→  for (let iter = 1; iter <= maxIter; iter++) {
   816→    const swarm = swarmAnalysis(root);
   817→    const test = scanTesting(root);
   818→    const quantum = quantumScore(swarm.findings, test, root);
   819→    const prevScore = history.length > 0 ? history[history.length - 1].score : 0;
   820→
   821→    // Auto-fix safe issues
   822→    let fixes = 0;
   823→    for (const issue of swarm.findings.filter(f => f.fixable)) {
   824→      try {
   825→        const fp = join(root, issue.file);
   826→        if (!existsSync(fp)) continue;
   827→        const content = readFileSync(fp, "utf-8");
   828→        const lines = content.split("\n");
   829→
   830→        if (issue.line && issue.line <= lines.length) {
   831→          const line = lines[issue.line - 1];
   832→          let fixed = false;
   833→
   834→          if (issue.message.includes("console.log")) {
   835→            lines[issue.line - 1] = line.replace(/console\.(log|debug)\s*\(/, "console.warn(");
   836→            fixed = true;
   837→          } else if (issue.message.includes("Empty catch")) {
   838→            const indent = line.match(/^(\s*)/)?.[1] || "";
   839→            lines.splice(issue.line, 0, `${indent}  console.error("Error:", err);`);
   840→            fixed = true;
   841→          } else if (issue.message.includes("innerHTML")) {
   842→            lines[issue.line - 1] = line.replace(/\.innerHTML\s*=/, ".textContent =");
   843→            fixed = true;
   844→          }
   845→
   846→          if (fixed) { writeFileSync(fp, lines.join("\n")); fixes++; }
   847→        }
   848→      } catch { /* Fix failed — continue */ }
   849→    }
   850→    totalFixes += fixes;
   851→
   852→    history.push({ iteration: iter, score: quantum.quantumScore, grade: quantum.grade, fixes, findings: swarm.findings.length, confidence: swarm.confidence });
   853→
   854→    if (quantum.quantumScore >= targetScore) break;
   855→    if (iter > 2 && history.slice(-3).every(h => h.score === history.slice(-3)[0].score)) break;
   856→  }
   857→
   858→  return { history, totalFixes, finalScore: history.length > 0 ? history[history.length - 1].score : 0 };
   859→}
