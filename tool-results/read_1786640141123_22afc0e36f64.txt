    97→// ═══════════════════════════════════════════════════════════════
    98→function WaveformBars({ playing, duration }: { playing: boolean; duration: number }) {
    99→  const bars = 24;
   100→  return (
   101→    <div className="flex items-center gap-[2px] h-8">
   102→      {Array.from({ length: bars }).map((_, i) => {
   103→        const h = 20 + Math.sin(i * 0.8) * 30 + Math.random() * 20;
   104→        return (
   105→          <motion.div
   106→            key={i}
   107→            className={`w-[3px] rounded-full ${playing ? 'bg-primary-foreground' : 'bg-foreground/60'}`}
   108→            animate={playing ? { scaleY: [1, 1.4, 0.8, 1.2, 1] } : {}}
   109→            transition={{ duration: 0.8 + i * 0.05, repeat: Infinity, ease: 'easeInOut' }}
   110→            style={{ height: `${h}%`, transformOrigin: 'bottom' }}
   111→          />
   112→        );
   113→      })}
   114→    </div>
   115→  );
   116→}
