   202→function timeAgo(dateStr: string): string {
   203→  try { return formatDistanceToNow(new Date(dateStr), { addSuffix: true }); } catch { return ''; }
   204→}

   206→function fmtDate(dateStr: string): string {
   207→  try { return format(new Date(dateStr), 'MMM d, yyyy h:mm a'); } catch { return dateStr; }
   208→}

   210→function fmtDateShort(dateStr: string): string {
   211→  try { return format(new Date(dateStr), 'MMM d'); } catch { return dateStr; }

   218→function truncate(str: string | null | undefined, len: number): string {
   219→  if (!str) return '';
