    98→// WAVEFORM BARS (for voice messages)
    99→// ═══════════════════════════════════════════════════════════════
   100→function WaveformBars({ playing, duration }: { playing: boolean; duration: number }) {
   101→  const bars = 24;
   102→  return (
   103→    <div className="flex items-center gap-[2px] h-8">
   104→      {Array.from({ length: bars }).map((_, i) => {
   105→        const h = 20 + Math.sin(i * 0.8) * 30 + Math.abs(Math.sin(i * 1.7 + duration * 0.1)) * 20;
   106→        return (
   107→          <motion.div
   108→            key={i}
   109→            className={`w-[3px] rounded-full ${playing ? 'bg-primary-foreground' : 'bg-foreground/60'}`}
   110→            animate={playing ? { scaleY: [1, 1.4, 0.8, 1.2, 1] } : {}}
   111→            transition={{ duration: 0.8 + i * 0.05, repeat: Infinity, ease: 'easeInOut' }}
   112→            style={{ height: `${h}%`, transformOrigin: 'bottom' }}
   113→          />
   114→        );
   115→      })}
   116→    </div>
   117→  );
   118→}

   123→function formatDuration(s: number): string {
   124→  const m = Math.floor(s / 60);
   125→  const sec = s % 60;
   126→  return `${m}:${sec.toString().padStart(2, '0')}`;
   127→}
   128→
   129→function timeAgo(dateStr: string): string {
   130→  try { return formatDistanceToNow(new Date(dateStr), { addSuffix: true }); }
   131→  catch { return ''; }
   132→}
   133→
   134→function parseJsonData<T>(str?: string | null): T | null {
   135→  if (!str) return null;
   136→  try { return JSON.parse(str) as T; }
   137→  catch { return null; }
   138→}

   713→  // #30 Message Stats
   714→  const getStats = () => {
   715→    const mine = chatMessages.filter((m) => m.senderId === currentUser?.id);
   716→    const theirs = chatMessages.filter((m) => m.senderId !== currentUser?.id);
   717→    const totalChars = chatMessages.reduce((sum, m) => sum + m.content.length, 0);
   718→    let avgRespTime = 'N/A';
   719→    if (chatMessages.length >= 2) {
   720→      const gaps: number[] = [];
   721→      for (let i = 1; i < chatMessages.length; i++) {
   722→        if (chatMessages[i].senderId !== chatMessages[i - 1].senderId) {
   723→          const t1 = new Date(chatMessages[i - 1].createdAt).getTime();
   724→          const t2 = new Date(chatMessages[i].createdAt).getTime();
   725→          if (t2 > t1) gaps.push(t2 - t1);
   726→        }
   727→      }
   728→      if (gaps.length > 0) {
   729→        const avgMs = gaps.reduce((a, b) => a + b, 0) / gaps.length;
   730→        const totalSec = Math.floor(avgMs / 1000);
   731→        const m = Math.floor(totalSec / 60);
   732→        const s = totalSec % 60;
   733→        avgRespTime = `${m}m ${s}s`;
   734→      }
   735→    }
   736→    return { total: chatMessages.length, mine: mine.length, theirs: theirs.length, totalChars, avgRespTime };
   737→  };
