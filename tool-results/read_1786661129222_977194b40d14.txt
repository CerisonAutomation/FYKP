   187→function timeAgo(dateStr: string): string {
   188→  try { return formatDistanceToNow(new Date(dateStr), { addSuffix: true }); } catch { return ''; }
   189→}
   190→
   191→function fmtDate(dateStr: string): string {
   192→  try { return format(new Date(dateStr), 'MMM d, yyyy h:mm a'); } catch { return dateStr; }
   193→}
   194→
   195→function fmtDateShort(dateStr: string): string {
   196→  try { return format(new Date(dateStr), 'MMM d'); } catch { return dateStr; }
   197→}

   199→function getAvatar(user: { avatar?: string | null } | null | undefined): string {

   203→function truncate(str: string | null | undefined, len: number): string {
   204→  if (!str) return '';
   205→  return str.length > len ? str.slice(0, len) + '...' : str;
   206→}

   208→function getLastSeenText(user: User): string {
   209→  if (user.online && user.showOnline) return 'Online now';
   210→  if (user.lastSeen) return `Last seen ${timeAgo(user.lastSeen)}`;
   211→  return 'Offline';
   212→}
