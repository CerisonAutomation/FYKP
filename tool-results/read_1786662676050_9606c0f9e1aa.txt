    98→// WAVEFORM BARS (for voice messages)
    99→// ═══════════════════════════════════════════════════════════════
   100→function WaveformBars({ playing, duration }: { playing: boolean; duration: number }) {
   101→  const bars = 24;
   102→  return (
   103→    <div className="flex items-center gap-[2px] h-8">
   104→      {Array.from({ length: bars }).map((_, i) => {
   105→        const h = 20 + Math.sin(i * 0.8) * 30 + Math.abs(Math.sin(i * 1.7 + duration * 0.1)) * 20;
   106→        return (
   107→          <motion.div
   108→            key={i}
   109→            className={`w-[3px] rounded-full ${playing ? 'bg-primary-foreground' : 'bg-foreground/60'}`}
   110→            animate={playing ? { scaleY: [1, 1.4, 0.8, 1.2, 1] } : {}}
   111→            transition={{ duration: 0.8 + i * 0.05, repeat: Infinity, ease: 'easeInOut' }}
   112→            style={{ height: `${h}%`, transformOrigin: 'bottom' }}
   113→          />
   114→        );
   115→      })}
   116→    </div>
   117→  );
   118→}

   689→    const totalChars = chatMessages.reduce((sum, m) => sum + m.content.length, 0);
   690→    let avgRespTime = 'N/A';
   691→    if (chatMessages.length >= 2) {
   692→      const gaps: number[] = [];
   693→      for (let i = 1; i < chatMessages.length; i++) {
   694→        if (chatMessages[i].senderId !== chatMessages[i - 1].senderId) {
   695→          const t1 = new Date(chatMessages[i - 1].createdAt).getTime();
   696→          const t2 = new Date(chatMessages[i].createdAt).getTime();
   697→          if (t2 > t1) gaps.push(t2 - t1);
   698→        }
   699→      }
   700→      if (gaps.length > 0) {
   701→        const avgMs = gaps.reduce((a, b) => a + b, 0) / gaps.length;
   702→        const totalSec = Math.floor(avgMs / 1000);
   703→        const m = Math.floor(totalSec / 60);
   704→        const s = totalSec % 60;
   705→        avgRespTime = `${m}m ${s}s`;
   706→      }
   707→    }
   708→    return { total: chatMessages.length, mine: mine.length, theirs: theirs.length, totalChars, avgRespTime };
