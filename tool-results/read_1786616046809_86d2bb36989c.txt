    58→// ─── Helpers ─────────────────────────────────────────────────────────────────
    59→function timeAgo(dateStr: string): string {
    60→  try {
    61→    return formatDistanceToNow(new Date(dateStr), { addSuffix: true });
    62→  } catch {
    63→    return '';
    64→  }
    65→}
    66→
    67→function formatEventDate(dateStr: string): string {
    68→  try {
    69→    return format(new Date(dateStr), 'MMM d, yyyy h:mm a');
    70→  } catch {
    71→    return dateStr;
    72→  }
    73→}
    74→
    75→function getAvatar(user: { avatar?: string | null } | null | undefined): string {
    76→  return user?.avatar || DEFAULT_AVATAR;
    77→}
    78→
    79→function truncate(str: string | null | undefined, len: number): string {
    80→  if (!str) return '';
    81→  return str.length > len ? str.slice(0, len) + '...' : str;
    82→}
    83→
    84→function getLastSeenText(user: User): string {
    85→  if (user.online && user.showOnline) return 'Online now';
    86→  if (user.lastSeen) return `Last seen ${timeAgo(user.lastSeen)}`;
    87→  return 'Offline';
    88→}
