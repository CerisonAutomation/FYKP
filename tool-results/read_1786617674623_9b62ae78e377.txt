    61→function timeAgo(dateStr: string): string {
    62→  try {
    63→    return formatDistanceToNow(new Date(dateStr), { addSuffix: true });
    64→  } catch {
    65→    return '';
    66→  }
    67→}
    68→
    69→function formatEventDate(dateStr: string): string {
    70→  try {
    71→    return format(new Date(dateStr), 'MMM d, yyyy h:mm a');
    72→  } catch {
    73→    return dateStr;
    74→  }
    75→}
    76→
    77→function getAvatar(user: { avatar?: string | null } | null | undefined): string {
    78→  return user?.avatar || DEFAULT_AVATAR;
    79→}
    80→
    81→function truncate(str: string | null | undefined, len: number): string {
    82→  if (!str) return '';
    83→  return str.length > len ? str.slice(0, len) + '...' : str;
    84→}

    86→function getLastSeenText(user: User): string {
    87→  if (user.online && user.showOnline) return 'Online now';
    88→  if (user.lastSeen) return `Last seen ${timeAgo(user.lastSeen)}`;
    89→  return 'Offline';
    90→}
