   217→   217→   717→   717→function quantumScore(findings: readonly Finding[], testResult: TestResult, root: string): QuantumResult {
   218→   218→   718→   718→  const sec = findings.filter(f => f.message.match(/secret|eval|innerHTML|SQL|injection|shell|traversal|crypto|tls|prototype/));
   219→   219→   719→   719→  const qual = findings.filter(f => f.message.match(/catch|console|suppress|eslint|TODO|magic|modularity|oversized/));

   225→   225→   725→   725→

   227→   227→   727→   727→  const dimensions: DimensionScore[] = [
   228→   228→   728→   728→    { name: "Security", score: Math.max(0, 100 - secCritical * 25 - secHigh * 10 - sec.length * 2), weight: 0.20 },
   229→   229→   729→   729→    { name: "Quality", score: Math.max(0, 100 - qualEmpty * 10 - qual.length), weight: 0.15 },
   230→   230→   730→   730→    { name: "Testing", score: Math.min(100, Math.round(testResult.ratio * 500)), weight: 0.15 },
   231→   231→   731→   731→    { name: "Architecture", score: Math.max(0, 100 - findings.filter(f => f.category === "nesting" || f.category === "modularity").length * 5), weight: 0.10 },
   232→   232→   732→   732→    { name: "Performance", score: 85, weight: 0.10 },
   233→   233→   733→   733→    { name: "Reliability", score: Math.max(0, 100 - qualEmpty * 8 - mem.length * 5), weight: 0.10 },
   234→   234→   734→   734→    { name: "Maintainability", score: Math.max(0, 100 - qual.filter(f => f.category === "modularity" || f.category === "magic-numbers" || f.category === "debt").length * 3), weight: 0.05 },
   235→   235→   735→   735→    { name: "Memory", score: Math.max(0, 100 - mem.length * 10), weight: 0.05 },
   236→   236→   736→   736→    { name: "Documentation", score: existsSync(join(root, "README.md")) ? 85 : 40, weight: 0.03 },
   237→   237→   737→   737→    { name: "Deployability", score: existsSync(join(root, ".github/workflows")) ? 90 : (existsSync(join(root, "package.json")) ? 60 : 30), weight: 0.02 },
   238→   238→   738→   738→  ];
   239→   239→   739→   739→

   247→   247→   747→   747→  // Weighted average with 1.5x multiplier for "exceeding expectations"

   249→   249→   749→   749→  const quantum = Math.round(raw * 1.5);
   250→   250→   750→   750→

   256→   256→   756→   756→  else if (quantum >= 50) grade = "D";
   257→   257→   757→   757→  else grade = "F";
   258→   258→   758→   758→
   259→   259→   759→   759→  return { quantumScore: quantum, grade, dimensions, totalIssues: findings.length };
   260→   260→   760→   760→}
