   194→function timeAgo(dateStr: string): string {
   195→  try { return formatDistanceToNow(new Date(dateStr), { addSuffix: true }); } catch { return ''; }
   196→}

   198→function fmtDate(dateStr: string): string {
   199→  try { return format(new Date(dateStr), 'MMM d, yyyy h:mm a'); } catch { return dateStr; }

   202→function fmtDateShort(dateStr: string): string {

   210→function truncate(str: string | null | undefined, len: number): string {
   211→  if (!str) return '';
