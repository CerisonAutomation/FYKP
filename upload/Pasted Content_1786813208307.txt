type LogLevel = "debug" | "info" | "warn" | "error";

interface LogEntry {
  readonly timestamp: string;
  readonly level: LogLevel;
  readonly component: string;
  readonly message: string;
  readonly data?: Record<string, unknown>;
}

class AuditLogger {
  private entries: LogEntry[] = [];
  private maxEntries = 1000;

  log(level: LogLevel, component: string, message: string, data?: Record<string, unknown>): void {
    const entry: LogEntry = { timestamp: new Date().toISOString(), level, component, message, data };
    this.entries.push(entry);
    if (this.entries.length > this.maxEntries) this.entries.splice(0, this.entries.length - 500);
    // Structured output to stderr (MCP convention)
    const prefix = level === "error" ? "❌" : level === "warn" ? "⚠️" : level === "info" ? "✅" : "🔍";
    console.error(`[repo-intel] ${prefix} [${component}] ${message}`);
  }

  debug(component: string, message: string, data?: Record<string, unknown>): void { this.log("debug", component, message, data); }
  info(component: string, message: string, data?: Record<string, unknown>): void { this.log("info", component, message, data); }
  warn(component: string, message: string, data?: Record<string, unknown>): void { this.log("warn", component, message, data); }
  error(component: string, message: string, data?: Record<string, unknown>): void { this.log("error", component, message, data); }

  getEntries(filter?: { level?: LogLevel; component?: string; limit?: number }): readonly LogEntry[] {
    let result = this.entries;
    if (filter?.level) result = result.filter(e => e.level === filter.level);
    if (filter?.component) result = result.filter(e => e.component === filter.component);
    if (filter?.limit) result = result.slice(-filter.limit);
    return result;
  }
}

const logger = new AuditLogger();

class CircuitBreaker {
  private failures = 0;
  private lastFailure = 0;
  private state: "closed" | "open" | "half-open" = "closed";
  private readonly threshold: number;
  private readonly resetTimeout: number;

  constructor(threshold = 5, resetTimeout = 60000) {
    this.threshold = threshold;
    this.resetTimeout = resetTimeout;
  }

  canExecute(): boolean {
    if (this.state === "closed") return true;
    if (this.state === "open" && Date.now() - this.lastFailure > this.resetTimeout) {
      this.state = "half-open";
      return true;
    }
    return this.state === "half-open";
  }

  recordSuccess(): void { this.failures = 0; this.state = "closed"; }
  recordFailure(): void { this.failures++; this.lastFailure = Date.now(); if (this.failures >= this.threshold) this.state = "open"; }
  getState(): string { return this.state; }
  reset(): void { this.failures = 0; this.state = "closed"; }
}

const circuitBreakers = new Map<string, CircuitBreaker>();
function getCircuitBreaker(name: string): CircuitBreaker {
  if (!circuitBreakers.has(name)) circuitBreakers.set(name, new CircuitBreaker());
  return circuitBreakers.get(name)!;
}

// ═══════════════════════════════════════════════════════════════════════════════
// SECTION 4: MEMORY-SAFE FILE WALKER
// ═══════════════════════════════════════════════════════════════════════════════

const EXCLUDE = new Set(["node_modules", ".git", ".next", "dist", "build", "__pycache__", "coverage", ".dyad", ".claude", ".github", ".repo-intel"]);
const MAX_FILE_SIZE = 1_000_000; // 1MB max per file
const MAX_TOTAL_BYTES = 50_000_000; // 50MB total scan limit

function walk(root: string, filter: RegExp, maxFiles: number = 500, maxDepth: number = 6): FileData[] {
  const files: FileData[] = [];
  let totalBytes = 0;

  function _walk(dir: string, depth: number): void {
    if (depth > maxDepth || files.length >= maxFiles || totalBytes >= MAX_TOTAL_BYTES) return;
    let entries: string[];
    try { entries = readdirSync(dir); } catch { return; }

    for (const entry of entries) {
      if (files.length >= maxFiles || totalBytes >= MAX_TOTAL_BYTES) return;
      if (EXCLUDE.has(entry) || entry.startsWith(".")) continue;

      const fp = join(dir, entry);
      let st: ReturnType<typeof statSync>;
      try { st = statSync(fp); } catch { continue; }

      if (st.isDirectory()) {
        _walk(fp, depth + 1);
      } else if (st.isFile() && st.size <= MAX_FILE_SIZE && filter.test(entry)) {
        try {
          const content = readFileSync(fp, "utf-8");
          totalBytes += content.length;
          const ext = entry.split(".").pop() || "unknown";
          files.push({ path: fp, rel: relative(root, fp), content, lines: Object.freeze(content.split("\n")), size: st.size, ext });
        } catch { /* Skip unreadable files */ }
      }
    }
  }

  _walk(root, 0);
  logger.info("walk", `Scanned ${files.length} files (${(totalBytes / 1024).toFixed(0)}KB)`);
  return files;
}

// SECTION 5: SANITIZED EXEC
// ═══════════════════════════════════════════════════════════════════════════════

const DANGEROUS_CHARS = /[;&|`$(){}[\]!#~\0]/;
const MAX_CMD_LENGTH = 1024;

function sanitizeCommand(cmd: string): string {
  // Remove null bytes, dangerous characters, and truncate
  return cmd.replace(/\0/g, "").replace(/[;&|`$(){}[\]!#~]/g, "").substring(0, MAX_CMD_LENGTH);
}

function safeExec(root: string, name: string, cmd: string, timeout = 60000): VerifyCheck {
  const cb = getCircuitBreaker(`exec:${name}`);
  if (!cb.canExecute()) {
    logger.warn("exec", `Circuit breaker open for ${name}, skipping`);
    return { name, status: "skip", output: "Circuit breaker open — too many recent failures" };
  }

  const sanitized = sanitizeCommand(cmd);
  try {
    execFileSync("sh", ["-c", sanitized], { cwd: root, timeout, stdio: "pipe", env: { ...process.env, NODE_OPTIONS: "" } });
    cb.recordSuccess();
    return { name, status: "pass", command: sanitized };
  } catch (err: any) {
    cb.recordFailure();
    const output = (err.stderr || err.stdout || err.message || "").substring(0, 500);
    logger.error("exec", `${name} failed`, { exitCode: err.status, output: output.substring(0, 200) });
    return { name, status: "fail", command: sanitized, output };
  }
}

let findingCounter = 0;
function createFinding(severity: Severity, file: string, line: number | undefined, message: string, category: Category, agent: Agent, fixable: boolean): Finding {
  return { id: `F${++findingCounter}`, severity, file, line, message, category, agent, fixable };
}

function isScannerCode(filePath: string): boolean {
  return SCANNER_KEYWORDS.has(filePath) || filePath.includes("repo-intel");
}

interface CritiqueResult {
  readonly validated: readonly Finding[];
  readonly falsePositives: number;
  readonly confidence: number;
  readonly deduplicated: number;
}

function critiqueFindings(findings: readonly Finding[]): CritiqueResult {
  const validated: Finding[] = [];
  let falsePositives = 0;
  let deduplicated = 0;
  const seen = new Set<string>();

  for (const f of findings) {
    // Skip test file false positives
    if (f.severity === "critical" && f.file.match(/\.(test|spec|mock|fixture)\./)) { falsePositives++; continue; }
    // Skip vendor code
    if (f.file.includes("vendor/")) { falsePositives++; continue; }
    // Deduplicate
    const key = `${f.file}:${f.line}:${f.message}`;
    if (seen.has(key)) { deduplicated++; continue; }
    seen.add(key);
    validated.push(f);
  }

  const confidence = findings.length > 0 ? Math.round((validated.length / findings.length) * 100) : 100;
  logger.info("critique", `Validated ${validated.length}/${findings.length} findings (FP: ${falsePositives}, Deduped: ${deduplicated}, Confidence: ${confidence}%)`);
  return { validated, falsePositives, confidence, deduplicated };
}

function scanSecurity(root: string, maxFiles: number = 500): readonly Finding[] {
  const cb = getCircuitBreaker("scanSecurity");
  if (!cb.canExecute()) { logger.warn("scanSecurity", "Circuit breaker open"); return []; }

  const findings: Finding[] = [];
  const files = walk(root, /\.(ts|tsx|js|jsx|py|go|rs|java|rb|php)$/, maxFiles);

  for (const f of files) {
    if (isScannerCode(f.rel)) continue; // Skip scanner code itself

    for (let i = 0; i < f.lines.length; i++) {
      const l = f.lines[i];

      // Hardcoded secrets (improved regex — covers more patterns)
      if (/(?:password|secret|api[_-]?key|token|credentials?|auth[_-]?key)\s*[:=]\s*["'`][A-Za-z0-9+/=_-]{16,}["'`]/i.test(l)) {
        findings.push(createFinding("critical", f.rel, i + 1, "Hardcoded secret/credential detected", "secrets", "security", false));
      }

      // Code injection (eval/exec/Function)
      if (/\b(?:eval|exec|Function|setTimeout|setInterval)\s*\(\s*["'`]/.test(l)) {
        findings.push(createFinding("critical", f.rel, i + 1, "Dynamic code execution from string literal", "injection", "security", false));
      }

      // innerHTML XSS
      if (/\.innerHTML\s*=/.test(l) && !/^\s*\/\//.test(l)) {
        findings.push(createFinding("high", f.rel, i + 1, "innerHTML assignment — XSS vulnerability", "xss", "security", true));
      }

      // SQL injection via template literals
      if (/`[^`]*\$\{[^}]*\}[^`]*(?:SELECT|INSERT|UPDATE|DELETE|DROP)/i.test(l)) {
        findings.push(createFinding("critical", f.rel, i + 1, "SQL injection via template literal interpolation", "injection", "security", false));
      }

      // Shell injection
      if (/(?:execSync|exec|spawn|system|child_process)\s*\(/.test(l) && /shell\s*[:=]\s*true/.test(l)) {
        findings.push(createFinding("high", f.rel, i + 1, "Shell injection risk — shell: true with user input", "injection", "security", false));
      }

      // Path traversal
      if (/(?:path\.join|path\.resolve)\s*\(.*(?:req|request|params|query|body|input)/.test(l)) {
        findings.push(createFinding("medium", f.rel, i + 1, "Potential path traversal — user input in path construction", "traversal", "security", true));
      }

      // Weak hashing
      if (/\b(?:md5|sha1)\s*\(/.test(l) && !/^\s*\/\//.test(l)) {
        findings.push(createFinding("medium", f.rel, i + 1, "Weak hashing algorithm (md5/sha1) — use sha256+ or bcrypt", "crypto", "security", true));
      }

      // TLS verification disabled
      if (/verify\s*[:=]\s*false/i.test(l) && !/^\s*\/\//.test(l)) {
        findings.push(createFinding("high", f.rel, i + 1, "TLS verification disabled — MITM risk", "tls", "security", true));
      }

      // Prototype pollution
      if (/__proto__|constructor\[/.test(l)) {
        findings.push(createFinding("critical", f.rel, i + 1, "Prototype pollution vector", "injection", "security", false));
      }
    }
  }

  cb.recordSuccess();
  logger.info("scanSecurity", `Found ${findings.length} security issues across ${files.length} files`);
  return findings;
}

function scanQuality(root: string, maxFiles: number = 500): readonly Finding[] {
  const cb = getCircuitBreaker("scanQuality");
  if (!cb.canExecute()) { logger.warn("scanQuality", "Circuit breaker open"); return []; }

  const findings: Finding[] = [];
  const files = walk(root, /\.(ts|tsx|js|jsx)$/, maxFiles);

  for (const f of files) {
    const isTest = /\.(test|spec)\./.test(f.rel);
    if (isScannerCode(f.rel)) continue;

    for (let i = 0; i < f.lines.length; i++) {
      const l = f.lines[i];

      // Empty catch blocks (improved detection)
      if (/catch\s*\(/.test(l)) {
        const nextBlock = f.lines.slice(i + 1, i + 5).join("");
        if (/^\s*\{\s*\}/.test(nextBlock) || /^\s*\{\s*\/\*.*\*\/\s*\}/.test(nextBlock) || /^\s*\{\s*\/\/.*\}/.test(nextBlock)) {
          findings.push(createFinding("high", f.rel, i + 1, "Empty catch block — errors silently swallowed", "error-handling", "quality", true));
        }
      }

      // console.log in production (not tests, not scanner code)
      if (!isTest && /console\.(log|debug|info)\s*\(/.test(l) && !/^\s*\/\//.test(l)) {
        findings.push(createFinding("low", f.rel, i + 1, "console.log/debug/info in production code", "logging", "quality", true));
        break; // One per file
      }

      // TypeScript suppression
      if (/@ts-ignore|@ts-expect-error|@ts-nocheck/.test(l)) {
        findings.push(createFinding("medium", f.rel, i + 1, "TypeScript suppression directive — hides type errors", "type-safety", "quality", false));
      }

      // ESLint disable
      if (/eslint-disable(?!\s*-next-line)/.test(l)) {
        findings.push(createFinding("low", f.rel, i + 1, "ESLint disable (broad suppression)", "linting", "quality", true));
      }

      // TODO/FIXME/HACK
      const todoMatch = l.match(/\/\/\s*(TODO|FIXME|HACK|XXX|TEMP)\b/i);
      if (todoMatch) {
        findings.push(createFinding("low", f.rel, i + 1, `Unresolved ${todoMatch[1]} comment — technical debt`, "debt", "quality", false));
      }

      // Magic numbers (heuristic: >4 digits, not in constants/imports)
      if (!/^\s*\/\//.test(l) && /\b\d{5,}\b/.test(l) && !/const|let|var|import|require|version|port|timeout|limit|copyright|20\d{2}/i.test(l)) {
        findings.push(createFinding("low", f.rel, i + 1, "Magic number — extract to named constant", "magic-numbers", "quality", true));
      }
    }

    // File-level checks
    if (f.lines.length > 500) {
      findings.push(createFinding("medium", f.rel, undefined, `Large file (${f.lines.length} lines) — consider splitting`, "modularity", "quality", false));
    }
    if (f.size > 50000) {
      findings.push(createFinding("medium", f.rel, undefined, `Oversized file (${(f.size / 1024).toFixed(0)}KB) — consider splitting`, "modularity", "quality", false));
    }
  }

  cb.recordSuccess();
  logger.info("scanQuality", `Found ${findings.length} quality issues across ${files.length} files`);
  return findings;
}

function scanArchitecture(root: string): readonly Finding[] {
  const cb = getCircuitBreaker("scanArchitecture");
  if (!cb.canExecute()) { logger.warn("scanArchitecture", "Circuit breaker open"); return []; }

  const findings: Finding[] = [];

  // Deep nesting detection
  let deepDirs = 0;
  function checkDepth(dir: string, d: number): void {
    if (d > 7) { deepDirs++; return; }
    let entries: string[];
    try { entries = readdirSync(dir); } catch { return; }
    for (const entry of entries) {
      if (EXCLUDE.has(entry) || entry.startsWith(".")) continue;
      const fp = join(dir, entry);
      try { if (statSync(fp).isDirectory()) checkDepth(fp, d + 1); } catch {}
    }
  }
  checkDepth(root, 0);
  if (deepDirs > 0) {
    findings.push(createFinding("medium", "(directory structure)", undefined, `${deepDirs} directories deeper than 7 levels — consider flattening`, "nesting", "architecture", false));
  }

  cb.recordSuccess();
  return findings;
}

function scanMemoryLeaks(root: string, maxFiles: number = 300): readonly Finding[] {
  const cb = getCircuitBreaker("scanMemoryLeaks");
  if (!cb.canExecute()) { logger.warn("scanMemoryLeaks", "Circuit breaker open"); return []; }

  const findings: Finding[] = [];
  const files = walk(root, /\.(ts|tsx|js|jsx)$/, maxFiles);

  for (const f of files) {
    if (isScannerCode(f.rel)) continue;

    for (let i = 0; i < f.lines.length; i++) {
      const l = f.lines[i];

      // addEventListener without removeEventListener
      if (/\.addEventListener\s*\(/.test(l)) {
        const hasRemove = f.lines.some((x, j) => j !== i && /\.removeEventListener\s*\(/.test(x));
        if (!hasRemove) {
          findings.push(createFinding("high", f.rel, i + 1, "addEventListener without matching removeEventListener", "memory-leak", "memory", true));
        }
      }

      // useEffect without cleanup return
      if (/useEffect\s*\(/.test(l)) {
        let depth = 0;
        let hasReturn = false;
        for (let j = i; j < Math.min(i + 60, f.lines.length); j++) {
          depth += (f.lines[j].match(/{/g) || []).length;
          depth -= (f.lines[j].match(/}/g) || []).length;
          if (/return\s*(?:\(|function|\()/m.test(f.lines[j])) hasReturn = true;
          if (depth <= 0 && j > i) break;
        }
        if (!hasReturn) {
          findings.push(createFinding("high", f.rel, i + 1, "useEffect without cleanup return — subscriptions/timers may leak", "memory-leak", "memory", true));
        }
      }
    }
  }

  cb.recordSuccess();
  return findings;
}

function scanCleanup(root: string, maxFiles: number = 300): readonly Finding[] {
  const cb = getCircuitBreaker("scanCleanup");
  if (!cb.canExecute()) { logger.warn("scanCleanup", "Circuit breaker open"); return []; }

  const findings: Finding[] = [];
  const files = walk(root, /\.(ts|tsx|js|jsx|json|md|yaml|yml|css|html)$/, maxFiles);

  for (const f of files) {
    if (f.content.includes("@generated") || f.content.includes("AUTO GENERATED") || f.content.includes("do not edit")) {
      findings.push(createFinding("medium", f.rel, undefined, "Auto-generated file in source tree", "cleanup", "cleanup", true));
    }
    if (f.content.trim().length === 0 || f.content.trim().match(/^(\/\/|#|\/\*|\*\/?\s*)*$/)) {
      findings.push(createFinding("low", f.rel, undefined, "Empty or comment-only file", "cleanup", "cleanup", true));
    }
    if (f.rel.match(/\.(bak|backup|old|orig|save|tmp)$/) && statSync(f.path).mtimeMs < Date.now() - 30 * 86400000) {
      findings.push(createFinding("low", f.rel, undefined, "Stale backup file (>30 days old)", "cleanup", "cleanup", true));
    }
    const todoCount = (f.content.match(/(?:TODO|FIXME|HACK|XXX)/gi) || []).length;
    if (todoCount > 5) {
      findings.push(createFinding("medium", f.rel, undefined, `${todoCount} TODO/FIXME comments — accumulated tech debt`, "debt", "cleanup", false));
    }
  }

  cb.recordSuccess();
  return findings;
}

function scanGaps(root: string): readonly Finding[] {
  const findings: Finding[] = [];

  if (!existsSync(join(root, "tsconfig.json"))) findings.push(createFinding("medium", "tsconfig.json", undefined, "No TypeScript configuration", "gap", "gap", true));
  if (!existsSync(join(root, ".eslintrc.js")) && !existsSync(join(root, "eslint.config.js"))) findings.push(createFinding("medium", ".eslintrc", undefined, "No linter configured", "gap", "gap", true));
  if (!existsSync(join(root, ".github/workflows"))) findings.push(createFinding("high", ".github/workflows", undefined, "No CI/CD pipeline", "gap", "gap", false));
  if (!existsSync(join(root, "README.md"))) findings.push(createFinding("high", "README.md", undefined, "No README documentation", "gap", "gap", true));
  if (!existsSync(join(root, ".gitignore"))) findings.push(createFinding("high", ".gitignore", undefined, "No .gitignore", "gap", "gap", true));
  if (existsSync(join(root, "package.json"))) {
    try {
      const pkg = JSON.parse(readFileSync(join(root, "package.json"), "utf-8"));
      if (!pkg.scripts?.test) findings.push(createFinding("medium", "package.json", undefined, "No test script configured", "gap", "gap", false));
      if (!pkg.scripts?.lint) findings.push(createFinding("low", "package.json", undefined, "No lint script configured", "gap", "gap", false));
    } catch { /* JSON parse error — skip */ }
  }

  return findings;
}

function scanTesting(root: string): TestResult {
  const findings: Finding[] = [];
  const testFiles = walk(root, /\.(test|spec)\.(ts|tsx|js|jsx)$/, 500);
  const sourceFiles = walk(root, /\.(ts|tsx|js|jsx)$/, 2000).filter(f => ! /\.(test|spec)\./.test(f.rel));
  const ratio = sourceFiles.length > 0 ? testFiles.length / sourceFiles.length : 0;

  for (const tf of testFiles) {
    const hasAssert = tf.lines.some(l => /(?:expect|assert|should|\.toBe|\.toEqual|\.toMatch|\.toThrow|\.toHaveBeenCalled|\.resolves|\.rejects)/.test(l));
    if (!hasAssert && tf.lines.length > 10) {
      findings.push(createFinding("high", tf.rel, undefined, "Test file has no assertions — passes by not throwing", "test-quality", "testing", false));
    }
  }

  for (const tf of testFiles) {
    for (const l of tf.lines) {
      if (/\b(xit|xtest|\.skip|pending|@ignore|@Disabled|\.only)\b/.test(l)) {
        findings.push(createFinding("medium", tf.rel, undefined, "Test is disabled/skipped/only", "test-coverage", "testing", false));
        break;
      }
    }
  }

  return { testFiles: testFiles.length, sourceFiles: sourceFiles.length, ratio, findings };
}

// SECTION 8: VERIFICATION ENGINE
// ═══════════════════════════════════════════════════════════════════════════════

function verifyAll(root: string): VerifyResult {
  const checks: VerifyCheck[] = [];
  let scripts: Record<string, string> = {};
  if (existsSync(join(root, "package.json"))) {
    try { scripts = JSON.parse(readFileSync(join(root, "package.json"), "utf-8")).scripts || {}; } catch {}
  }

  if (scripts.build) checks.push(safeExec(root, "build", "npm run build 2>&1 | tail -5"));
  else if (existsSync(join(root, "tsconfig.json"))) checks.push(safeExec(root, "typecheck", "npx tsc --noEmit 2>&1 | head -10"));
  else checks.push({ name: "build", status: "skip", output: "No build script or tsconfig" });

  if (scripts.lint) checks.push(safeExec(root, "lint", "npm run lint 2>&1 | head -5"));
  else checks.push({ name: "lint", status: "skip", output: "No lint script" });

  if (scripts.test) checks.push(safeExec(root, "test", "npm test 2>&1 | tail -5"));
  else checks.push({ name: "test", status: "skip", output: "No test script" });

  checks.push(safeExec(root, "secrets", 'grep -rn "sk-\\|api_key\\|password=" --include="*.ts" --include="*.js" . 2>/dev/null | grep -v node_modules | head -5'));

  const passed = checks.filter(c => c.status === "pass").length;
  const failed = checks.filter(c => c.status === "fail").length;
  return { overall: failed === 0 ? "pass" : "fail", checks, summary: `${passed} passed, ${failed} failed` };
}

// ═══════════════════════════════════════════════════════════════════════════════
// SECTION 9: QUANTUM SCORING (Mathematically Sound)
// ═══════════════════════════════════════════════════════════════════════════════

function quantumScore(findings: readonly Finding[], testResult: TestResult, root: string): QuantumResult {
  const sec = findings.filter(f => f.message.match(/secret|eval|innerHTML|SQL|injection|shell|traversal|crypto|tls|prototype/));
  const qual = findings.filter(f => f.message.match(/catch|console|suppress|eslint|TODO|magic|modularity|oversized/));
  const mem = findings.filter(f => f.message.match(/addEventListener|useEffect|cleanup|memory/));

  const secCritical = sec.filter(f => f.severity === "critical").length;
  const secHigh = sec.filter(f => f.severity === "high").length;
  const qualEmpty = qual.filter(f => f.message.includes("catch")).length;

  // Mathematically sound weighting: weights sum to 1.0
  const dimensions: DimensionScore[] = [
    { name: "Security", score: Math.max(0, 100 - secCritical * 25 - secHigh * 10 - sec.length * 2), weight: 0.20 },
    { name: "Quality", score: Math.max(0, 100 - qualEmpty * 10 - qual.length), weight: 0.15 },
    { name: "Testing", score: Math.min(100, Math.round(testResult.ratio * 500)), weight: 0.15 },
    { name: "Architecture", score: Math.max(0, 100 - findings.filter(f => f.category === "nesting" || f.category === "modularity").length * 5), weight: 0.10 },
    { name: "Performance", score: 85, weight: 0.10 },
    { name: "Reliability", score: Math.max(0, 100 - qualEmpty * 8 - mem.length * 5), weight: 0.10 },
    { name: "Maintainability", score: Math.max(0, 100 - qual.filter(f => f.category === "modularity" || f.category === "magic-numbers" || f.category === "debt").length * 3), weight: 0.05 },
    { name: "Memory", score: Math.max(0, 100 - mem.length * 10), weight: 0.05 },
    { name: "Documentation", score: existsSync(join(root, "README.md")) ? 85 : 40, weight: 0.03 },
    { name: "Deployability", score: existsSync(join(root, ".github/workflows")) ? 90 : (existsSync(join(root, "package.json")) ? 60 : 30), weight: 0.02 },
  ];

  // Verify weights sum to 1.0
  const weightSum = dimensions.reduce((s, d) => s + d.weight, 0);
  if (Math.abs(weightSum - 1.0) > 0.001) {
    logger.warn("quantumScore", `Weights sum to ${weightSum}, expected 1.0 — normalizing`);
    dimensions.forEach(d => { d.weight = d.weight / weightSum; });
  }

  // Weighted average with 1.5x multiplier for "exceeding expectations"
  const raw = dimensions.reduce((s, d) => s + d.score * d.weight, 0);
  const quantum = Math.round(raw * 1.5);

  let grade: Grade;
  if (quantum >= 150) grade = "S";
  else if (quantum >= 120) grade = "A";
  else if (quantum >= 90) grade = "B";
  else if (quantum >= 70) grade = "C";
  else if (quantum >= 50) grade = "D";
  else grade = "F";

  return { quantumScore: quantum, grade, dimensions, totalIssues: findings.length };
}

// SECTION 10: SWARM ANALYSIS
// ═══════════════════════════════════════════════════════════════════════════════

function swarmAnalysis(root: string): SwarmResult {
  const sec = scanSecurity(root, 300);
  const qual = scanQuality(root, 300);
  const arch = scanArchitecture(root);
  const mem = scanMemoryLeaks(root, 200);
  const clean = scanCleanup(root, 200);
  const gaps = scanGaps(root);
  const test = scanTesting(root);

  const all = [...sec, ...qual, ...arch, ...mem, ...clean, ...gaps, ...test.findings];
  const { validated, falsePositives, confidence } = critiqueFindings(all);

  const byAgent: Record<string, number> = {};
  const bySeverity: Record<string, number> = {};
  for (const f of validated) {
    byAgent[f.agent] = (byAgent[f.agent] ?? 0) + 1;
    bySeverity[f.severity] = (bySeverity[f.severity] ?? 0) + 1;
  }

  // Cross-agent correlations
  const correlations: string[] = [];
  const secFiles = new Set(sec.map(f => f.file));
  const memFiles = new Set(mem.map(f => f.file));
  const overlap = [...secFiles].filter(f => memFiles.has(f));
  if (overlap.length > 0) correlations.push(`${overlap.length} files have both security and memory issues`);

  return { findings: validated, falsePositives, confidence, byAgent, bySeverity, correlations, testResult: test };
}

function dynamicPersona(findings: readonly Finding[], scores: Record<string, number>): { recommended: string; weakest: { dimension: string; score: number }; strongest: { dimension: string; score: number }; rationale: string } {
  const dims = Object.entries(scores).sort((a, b) => a[1] - b[1]);
  const weakest = dims[0]; const strongest = dims[dims.length - 1];
  const personaMap: Record<string, string> = { security: "Security Engineer", quality: "Senior Developer", testing: "QA Lead", architecture: "CTO", reliability: "SRE", maintainability: "Senior Developer", memory: "SRE", documentation: "Product Manager", deployability: "SRE" };
  const recommended = personaMap[weakest[0]] || "Senior Developer";
  return { recommended, weakest: { dimension: weakest[0], score: weakest[1] }, strongest: { dimension: strongest[0], score: strongest[1] }, rationale: `Weakest area: ${weakest[0]} (${weakest[1]}/100). ${recommended} persona focuses on this area.` };
}

// SECTION 12: AUTO-DIAGNOSE LOOP
// ═══════════════════════════════════════════════════════════════════════════════

function autoDiagnose(root: string, targetScore: number = 85, maxIter: number = 10): { history: readonly { iteration: number; score: number; grade: Grade; fixes: number; findings: number; confidence: number }[]; totalFixes: number; finalScore: number } {
  const history: Array<{ iteration: number; score: number; grade: Grade; fixes: number; findings: number; confidence: number }> = [];
  let totalFixes = 0;

  for (let iter = 1; iter <= maxIter; iter++) {
    const swarm = swarmAnalysis(root);
    const test = scanTesting(root);
    const quantum = quantumScore(swarm.findings, test, root);
    const prevScore = history.length > 0 ? history[history.length - 1].score : 0;

    // Auto-fix safe issues
    let fixes = 0;
    for (const issue of swarm.findings.filter(f => f.fixable)) {
      try {
        const fp = join(root, issue.file);
        if (!existsSync(fp)) continue;
        const content = readFileSync(fp, "utf-8");
        const lines = content.split("\n");

        if (issue.line && issue.line <= lines.length) {
          const line = lines[issue.line - 1];
          let fixed = false;

          if (issue.message.includes("console.log")) {
            lines[issue.line - 1] = line.replace(/console\.(log|debug)\s*\(/, "console.warn(");
            fixed = true;
          } else if (issue.message.includes("Empty catch")) {
            const indent = line.match(/^(\s*)/)?.[1] || "";
            lines.splice(issue.line, 0, `${indent}  console.error("Error:", err);`);
            fixed = true;
          } else if (issue.message.includes("innerHTML")) {
            lines[issue.line - 1] = line.replace(/\.innerHTML\s*=/, ".textContent =");
            fixed = true;
          }

          if (fixed) { writeFileSync(fp, lines.join("\n")); fixes++; }
        }
      } catch { /* Fix failed — continue */ }
    }
    totalFixes += fixes;

    history.push({ iteration: iter, score: quantum.quantumScore, grade: quantum.grade, fixes, findings: swarm.findings.length, confidence: swarm.confidence });

    if (quantum.quantumScore >= targetScore) break;
    if (iter > 2 && history.slice(-3).every(h => h.score === history.slice(-3)[0].score)) break;
  }

  return { history, totalFixes, finalScore: history.length > 0 ? history[history.length - 1].score : 0 };
}

  const files = walk(path, /\.(ts|tsx|js|jsx)$/, 300);

  // Find the entry point
  const entryFile = files.find(f => f.rel.includes(entryPoint) || f.rel.endsWith(entryPoint));
  if (!entryFile) {
    return { content: [{ type: "text", text: JSON.stringify({ entryPoint, error: "Entry point not found", suggestion: "Try a file path or function name" }, null, 2) }], isError: true };
  }

  // Trace dependencies recursively
  const traced = new Set<string>();
  const callChain: Array<{ from: string; to: string; line: number }> = [];

  function trace(file: FileData, currentDepth: number): void {
    if (currentDepth >= depth || traced.has(file.rel)) return;
    traced.add(file.rel);

    for (let i = 0; i < file.lines.length; i++) {
      const importMatch = file.lines[i].match(/(?:import|from)\s+.*["']([^"']+)["']/);

        const targetFile = files.find(f => f.rel.includes(importMatch[1].replace(/\.(ts|tsx|js|jsx)$/, "")));
        if (targetFile) {
          callChain.push({ from: file.rel, to: targetFile.rel, line: i + 1 });
          trace(targetFile, currentDepth + 1);
        }
      }
    }
  }

  trace(entryFile, 0);

  const out = { entryPoint: entryFile.rel, depth, filesTraced: traced.size, callChain: callChain.slice(0, 50), summary: `Traced ${traced.size} files from ${entryFile.rel} with ${callChain.length} import relationships` };
